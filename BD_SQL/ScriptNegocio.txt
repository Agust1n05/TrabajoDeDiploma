/* ============================================================
   CREACIÓN DE TABLAS - NEGOCIO
   SQL SERVER
   ============================================================ */


/* ============================================================
   1. CLIENTE
   ============================================================ */

CREATE TABLE Cliente
(
    dni VARCHAR(50) NOT NULL,
    Nombre VARCHAR(50) NOT NULL,
    Apellido VARCHAR(50) NOT NULL,
    Telefono VARCHAR(50) NOT NULL,

    CONSTRAINT PK_Cliente
        PRIMARY KEY (dni)
);


/* ============================================================
   2. TECNICO
   ============================================================ */

CREATE TABLE Tecnico
(
    Dni VARCHAR(50) NOT NULL,
    Nombre VARCHAR(50) NOT NULL,
    Apellido VARCHAR(50) NOT NULL,
    Disponible BIT NOT NULL,

    CONSTRAINT PK_Tecnico
        PRIMARY KEY (Dni)
);


/* ============================================================
   3. EQUIPO
   ============================================================ */

CREATE TABLE Equipo
(
    idEquipo VARCHAR(50) NOT NULL,
    marca VARCHAR(50) NOT NULL,
    modelo VARCHAR(50) NOT NULL,
    imei VARCHAR(50) NOT NULL,
    descripcion VARCHAR(50) NOT NULL,

    CONSTRAINT PK_Equipo
        PRIMARY KEY (idEquipo)
);


/* ============================================================
   4. REPUESTO
   ============================================================ */

CREATE TABLE Repuesto
(
    Codigo VARCHAR(50) NOT NULL,
    stockDisponible INT NOT NULL,

    CONSTRAINT PK_Repuesto
        PRIMARY KEY (Codigo)
);


/* ============================================================
   5. ORDEN DE TRABAJO
   ============================================================ */

CREATE TABLE OrdenDeTrabajo
(
    NroOT INT IDENTITY(1,1) NOT NULL,
    Estado VARCHAR(MAX) NOT NULL,
    dni VARCHAR(50) NOT NULL,
    idEquipo VARCHAR(50) NOT NULL,
    DniTecnico VARCHAR(50) NOT NULL,

    CONSTRAINT PK_OrdenDeTrabajo
        PRIMARY KEY (NroOT),

    CONSTRAINT FK_OrdenDeTrabajo_Cliente
        FOREIGN KEY (dni)
        REFERENCES Cliente(dni),

    CONSTRAINT FK_OrdenDeTrabajo_Equipo
        FOREIGN KEY (idEquipo)
        REFERENCES Equipo(idEquipo),

    CONSTRAINT FK_OrdenDeTrabajo_Tecnico
        FOREIGN KEY (DniTecnico)
        REFERENCES Tecnico(Dni)
);


/* ============================================================
   6. DIAGNOSTICO
   ============================================================ */

CREATE TABLE Diagnostico
(
    IdDiagnostico INT IDENTITY(1,1) NOT NULL,
    detalle VARCHAR(50) NOT NULL,
    costo_mano_obra VARCHAR(50) NOT NULL,
    NroOT INT NOT NULL,

    CONSTRAINT PK_Diagnostico
        PRIMARY KEY (IdDiagnostico),

    CONSTRAINT FK_Diagnostico_OrdenDeTrabajo
        FOREIGN KEY (NroOT)
        REFERENCES OrdenDeTrabajo(NroOT)
);


/* ============================================================
   7. PRESUPUESTO
   ============================================================ */

CREATE TABLE Presupuesto
(
    IdPresupuesto INT IDENTITY(1,1) NOT NULL,
    TotalPresupuesto DECIMAL(18) NOT NULL,
    EstadoPresupuesto VARCHAR(50) NOT NULL,
    FechaRespuesta DATE NOT NULL,
    NroOT INT NOT NULL,

    CONSTRAINT PK_Presupuesto
        PRIMARY KEY (IdPresupuesto),

    CONSTRAINT FK_Presupuesto_OrdenDeTrabajo
        FOREIGN KEY (NroOT)
        REFERENCES OrdenDeTrabajo(NroOT)
);


/* ============================================================
   8. GARANTIA
   ============================================================ */

CREATE TABLE Garantia
(
    FechaVencimiento VARCHAR(50) NOT NULL,
    NroOT INT NOT NULL,

    CONSTRAINT FK_Garantia_OrdenDeTrabajo
        FOREIGN KEY (NroOT)
        REFERENCES OrdenDeTrabajo(NroOT)
);


/* ============================================================
   9. DETALLE REPUESTO
   ============================================================ */

CREATE TABLE DetalleRepuesto
(
    cantidadRequerida INT NOT NULL,
    CantidadUtilizada INT NOT NULL,
    Codigo VARCHAR(50) NOT NULL,
    IdDiagnostico INT NOT NULL,

    CONSTRAINT PK_DetalleRepuesto
        PRIMARY KEY (Codigo, IdDiagnostico),

    CONSTRAINT FK_DetalleRepuesto_Repuesto
        FOREIGN KEY (Codigo)
        REFERENCES Repuesto(Codigo),

    CONSTRAINT FK_DetalleRepuesto_Diagnostico
        FOREIGN KEY (IdDiagnostico)
        REFERENCES Diagnostico(IdDiagnostico)
);
